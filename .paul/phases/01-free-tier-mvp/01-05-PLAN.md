    ```
    // audit-added: sanitize cell content to prevent formula injection in Excel/LibreOffice.
    // Cells beginning with =, +, -, @ are treated as formulas by spreadsheet apps.
    function sanitizeCell(value: string): string {
      const escaped = value.replace(/"/g, '""');
      return /^[=+\-@\t\r]/.test(escaped) ? " " + escaped : escaped;
    }

    function handleDownloadCSV(data: AnalysisResponse, narrative: string | null) {
      const rows: string[][] = [
        ["Type", "Detail", "Sentiment Score"],
        ["Summary", "Overall Score", data.overall_score.toFixed(4)],
        ["Summary", "Total Reviews", String(data.total_reviews)],
        ["Summary", "Positive Reviews", String(data.distribution.positive)],
        ["Summary", "Neutral Reviews", String(data.distribution.neutral)],
        ["Summary", "Negative Reviews", String(data.distribution.negative)],
      ];
      if (narrative) {
        rows.push(["Narrative", sanitizeCell(narrative), ""]);
      }
      data.top_reviews.forEach((r) =>
        rows.push(["Positive Review", sanitizeCell(r.text), r.score.toFixed(4)])
      );
      data.bottom_reviews.forEach((r) =>
        rows.push(["Critical Review", sanitizeCell(r.text), r.score.toFixed(4)])
      );
      const csv = rows.map((row) => row.map((cell) => `"${cell}"`).join(",")).join("\n");
      // audit-added: prepend UTF-8 BOM (\uFEFF) so Windows Excel opens without
      // character corruption. Matches the BOM-handling requirement from plan 01-02.
      const blob = new Blob(["\uFEFF" + csv], { type: "text/csv;charset=utf-8;" });
      const url = URL.createObjectURL(blob);
      const link = document.createElement("a");
      link.href = url;
      link.download = "review-radar-report.csv";
      // audit-added: append to DOM before click (required for Safari) and defer
      // revokeObjectURL to ensure blob URL is not expired before download queues.
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      setTimeout(() => URL.revokeObjectURL(url), 100);
    }
    ```
    Call from an onClick handler on the "Download CSV" button. Pass `data` and `narrative` props.

  <name>Task 3: Create Render deployment config</name>
  <files>render.yaml</files>
  <action>
    Create `render.yaml` at the repo root (not inside backend/).

    ```yaml
    # Review Radar — Render deployment config
    # Deploy this repo as a Render web service.
    #
    # Required env vars (set in Render dashboard — NOT in this file):
    #   ANTHROPIC_API_KEY  — Anthropic API key for Claude Haiku narrative
    #   ALLOWED_ORIGINS    — Comma-separated allowed origins, e.g.:
    #                        https://your-site.netlify.app,http://localhost:3000

    services:
      - type: web
        name: review-radar-api
        runtime: python
        rootDir: backend
        buildCommand: pip install -r requirements.txt
        startCommand: uvicorn main:app --host 0.0.0.0 --port $PORT
        # audit-added: reference the existing GET /health endpoint for application-level
        # health monitoring. Without this, Render only checks TCP port binding and will
        # not restart a process that has crashed-but-not-exited.
        healthCheckPath: /health
        # audit-added: Free tier spins down after 15 min idle → 30-60s cold start on
        # next request. This directly conflicts with the "insights in under 30 seconds"
        # product promise. Document as known limitation; upgrade plan to eliminate.
        envVars:
          - key: PYTHON_VERSION
            value: "3.11.0"
          - key: ANTHROPIC_API_KEY
            sync: false
          - key: ALLOWED_ORIGINS
            sync: false
    ```

    Notes:
    - `sync: false` means these values are NOT read from render.yaml — they must be set
      manually in the Render dashboard, keeping secrets out of version control.
    - `rootDir: backend` means Render runs buildCommand and startCommand relative to
      the `backend/` directory — `main.py` is at the root of that directory.
    - `$PORT` is injected by Render automatically; uvicorn must bind to it.
    - `PYTHON_VERSION: "3.11.0"` matches `backend/runtime.txt`.

    Avoid: Putting actual API keys or URLs in render.yaml.
  </action>
